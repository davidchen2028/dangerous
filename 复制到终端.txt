# 极危行动 — 复制到终端（一步一步来）

# 【第 1 步】进入文件夹
cd "/Users/admin/project/david/极危/极危行动"

# 【第 2 步】启动服务器（必须看到下面方框里的字）
./run.sh

# 正常时应显示：
#   ======================================
#     极危行动 — 联机大厅
#   ======================================
#     本机:   http://127.0.0.1:8080
#   ……
# 然后光标停住不动 = 成功，不要关这个窗口

# 【第 3 步】打开浏览器，地址栏输入：
http://localhost:8080

# ── 若提示「端口 8080 已被占用」──
# 说明服务器已经在跑了！不用再 run.sh，直接浏览器打开：
http://localhost:8080

# 若要关掉旧服务器再重启，先执行（把 PID 换成你终端里显示的数字）：
# kill 52424
# 然后再 ./run.sh

# 换端口启动（现在 run.sh 会认 PORT 变量）：
# PORT=8082 ./run.sh
# 浏览器打开 http://localhost:8082

# ── 若提示 permission denied ──
chmod +x run.sh get-ip.sh
./run.sh

# ── 若完全没有输出 ──
# 确认复制的是整行 cd ... 和 ./run.sh，并在「终端」里运行，不是网页里

# ── 手机怎么用（与电脑通用，同一 WiFi）──
# 1. 电脑先 ./run.sh 不要关
# 2. 电脑执行 ./get-ip.sh 看 IP
# 3. 手机 Safari/Chrome 打开 http://电脑IP:8080（端口与终端一致）
# 4. 房间号与电脑填相同，如 main
# 不要用手机「文件」打开 index.html
#
# ── 手机联机失败 ──
# 点「房间」→「进入」，看红色提示；成功时显示「已联机 · 房间 xxx」
